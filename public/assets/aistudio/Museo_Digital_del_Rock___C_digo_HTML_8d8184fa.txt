        <h3>Investigación, archivo y memoria rockera en San Cristóbal Sur</h3>
        <p>
          El <strong>Museo Digital del Rock de San Cristóbal Sur</strong> nace como un proyecto de trabajo de grado de la <strong>Licenciatura en Artes Visuales de la Universidad Pedagógica Nacional</strong>, orientado a investigar cómo la conservación, organización y puesta en circulación de archivos gráficos y sonoros permite acercar, visibilizar y difundir la historia del rock y el metal en la localidad.
        </p>
        <p>
          Lejos de entender el archivo como un depósito estático, esta propuesta concibe los afiches, fotografías de conciertos, convocatorias comunitarias, producciones musicales y testimonios orales como soportes vivos de identidad territorial. A través de este entorno digital, se busca articular las memorias de bandas, gestores culturales y espacios autogestionados para reconocer la escena rockera de San Cristóbal como patrimonio cultural local.
        </p>
      </div>

      <div class="grid-3">

        </div>
        <div class="card">

        </div>
        <div class="card">

  <script>
    // Desplegar / cerrar menú de tres punticos
    function toggleMenu() {
      const nav = document.getElementById('dropdownNav');
      nav.classList.toggle('open');
    }

    // Cerrar menú al hacer clic fuera
    document.addEventListener('click', function (e) {
      const menuWrapper = document.querySelector('.menu-wrapper');
      const nav = document.getElementById('dropdownNav');
      if (!menuWrapper.contains(e.target)) {
        nav.classList.remove('open');
      }
    });

    // Navegación entre secciones principales
    function navigateTo(sectionId) {
      document.querySelectorAll('.page-section').forEach(sec => {
        sec.classList.remove('active');
      });
      const target = document.getElementById(sectionId);
      if (target) {
        target.classList.add('active');
      }

      document.querySelectorAll('.nav-item').forEach(btn => {
        btn.classList.remove('active');
        if (btn.getAttribute('data-target') === sectionId) {
          btn.classList.add('active');
        }
      });

      document.getElementById('dropdownNav').classList.remove('open');
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    // Cambio de pestañas en Archivos (Gráficos / Sonoros)
    function switchArchiveTab(tabName) {
      document.getElementById('btnTabGraficos').classList.remove('active');
      document.getElementById('btnTabSonoros').classList.remove('active');
      document.getElementById('tab-graficos').classList.remove('active');
      document.getElementById('tab-sonoros').classList.remove('active');

      if (tabName === 'graficos') {
        document.getElementById('btnTabGraficos').classList.add('active');
        document.getElementById('tab-graficos').classList.add('active');
      } else {
        document.getElementById('btnTabSonoros').classList.add('active');
        document.getElementById('tab-sonoros').classList.add('active');
      }
    }

    // Buscador de bandas por nombre
    function filterBands() {
      const query = document.getElementById('bandSearchInput').value.toLowerCase().trim();
      const cards = document.querySelectorAll('.band-card');
      cards.forEach(card => {
        const name = card.getAttribute('data-name').toLowerCase();
        card.style.display = name.includes(query) ? 'flex' : 'none';
      });
    }
  </script>
